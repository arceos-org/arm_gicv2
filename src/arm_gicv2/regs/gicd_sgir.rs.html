1//! Software Generated Interrupt Register, GICD_SGIR
2//! The GICD_SGIR characteristics are:
3//!
4//! ## Purpose
5//!
6//! Controls the generation of SGIs.
7//!
8//! ## Usage constraints
9//!
10//! It is IMPLEMENTATION DEFINED whether the GICD_SGIR has any effect when the forwarding of interrupts by Distributor is disabled by
11//! the GICD_CTLR settings.
12//!
13//! ## Configurations
14//!
15//! This register is available in all configurations of the GIC. If the GIC implements the Security Extensions this register is Common.
16//!
17//! The NSATT field, bit [15], is implemented only if the GIC implements the Security Extensions.
18
19use tock_registers::register_bitfields;
20use tock_registers::registers::WriteOnly;
21
22register_bitfields! {u32,
23    pub GICD_SGIR [
24        /// Bits [31:26] Reserved.
25        Reserved31 OFFSET(26) NUMBITS(6) [],
26        /// [25:24] TargetListFilter
27        /// Determines how the distributor must process the requested SGI:
28        /// - 0b00 Forward the interrupt to the CPU interfaces specified in the CPUTargetList fielda.
29        /// - 0b01 Forward the interrupt to all CPU interfaces except that of the processor that requested the interrupt.
30        /// - 0b10 Forward the interrupt only to the CPU interface of the processor that requested the interrupt.
31        /// - 0b11 Reserved.
32        TargetListFilter OFFSET(24) NUMBITS(2) [
33            ForwardToCPUTargetList = 0b00,
34            ForwardToAllExceptRequester = 0b01,
35            ForwardToRequester = 0b10,
36            Reserved = 0b11
37        ],
38        /// [23:16] CPUTargetList
39        /// When TargetList Filter = 0b00, defines the CPU interfaces to which the Distributor must forward the interrupt.
40        /// Each bit of CPUTargetList[7:0] refers to the corresponding CPU interface, for example CPUTargetList[0] corresponds to CPU interface 0.
41        /// Setting a bit to 1 indicates that the interrupt must be forwarded to the corresponding interface.
42        /// If this field is 0x00 when TargetListFilter is 0b00, the Distributor does not forward the interrupt to any CPU interface.
43        CPUTargetList OFFSET(16) NUMBITS(8) [],
44        /// [15] NSATT
45        /// Implemented only if the GIC includes the Security Extensions.
46        /// Specifies the required security value of the SGI:
47        /// - 0 Forward the SGI specified in the SGIINTID field to a specified CPU interface only if the SGI is configured as Group 0 on that interface.
48        /// - 1 Forward the SGI specified in the SGIINTID field to a specified CPU interfaces only if the SGI is configured as Group 1 on that interface.
49        ///
50        /// This field is writable only by a Secure access. Any Non-secure write to the GICD_SGIR generates an SGI only if the specified SGI is programmed
51        /// as Group 1, regardless of the value of bit[15] of the write.
52        NSATT OFFSET(15) NUMBITS(1) [],
53        /// [14:4] - Reserved, SBZ.
54        Reserved14_4 OFFSET(4) NUMBITS(11) [],
55        /// [3:0] SGIINTID
56        /// The Interrupt ID of the SGI to forward to the specified CPU interfaces.
57        /// The value of this field is the Interrupt ID, in the range 0-15,
58        /// for example a value of 0b0011 specifies Interrupt ID 3.
59        SGIINTID OFFSET(0) NUMBITS(4) []
60    ]
61}
